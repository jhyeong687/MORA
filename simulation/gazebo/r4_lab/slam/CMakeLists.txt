cmake_minimum_required(VERSION 3.16)
project(r4_slam_motors LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(r4_motors SHARED MotorSystem.cc)
target_compile_features(r4_motors PRIVATE cxx_std_17)
target_link_libraries(r4_motors PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(r4_motors PRIVATE -Wall -Wextra)
